cmake_minimum_required(VERSION 3.20)
project(ArborEnumLiveBuilder LANGUAGES CXX)
find_package(Threads REQUIRED)

set(ARBORENUM_EMBED_UI "" CACHE FILEPATH "Single-file builder HTML to embed")
option(ARBORENUM_DESKTOP "Open the embedded UI in a local browser by default" OFF)
option(ARBORENUM_STATIC_RUNTIME "Link Linux builds statically (musl recommended)" OFF)
if(ARBORENUM_DESKTOP AND NOT ARBORENUM_EMBED_UI)
  message(FATAL_ERROR "Desktop builds require ARBORENUM_EMBED_UI")
endif()

function(configure_native target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /bigobj /utf-8 /permissive- /Zc:__cplusplus)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter)
  endif()
  if(WIN32)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(${target} PRIVATE ws2_32 psapi)
  endif()
  if(ARBORENUM_STATIC_RUNTIME AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(${target} PRIVATE -static)
  endif()
endfunction()

add_executable(arborenum-builder main.cpp)
configure_native(arborenum-builder)
if(WIN32)
  target_link_libraries(arborenum-builder PRIVATE shell32 bcrypt)
endif()
if(ARBORENUM_EMBED_UI)
  get_filename_component(ui_file "${ARBORENUM_EMBED_UI}" ABSOLUTE)
  set(ui_header "${CMAKE_CURRENT_BINARY_DIR}/embedded_ui.hpp")
  add_custom_command(OUTPUT "${ui_header}"
    COMMAND "${CMAKE_COMMAND}" "-DINPUT=${ui_file}" "-DOUTPUT=${ui_header}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_ui.cmake"
    DEPENDS "${ui_file}" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_ui.cmake"
    VERBATIM)
  target_sources(arborenum-builder PRIVATE "${ui_header}")
  target_include_directories(arborenum-builder PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(arborenum-builder PRIVATE ARBORENUM_EMBEDDED_UI)
endif()
if(ARBORENUM_DESKTOP)
  target_compile_definitions(arborenum-builder PRIVATE ARBORENUM_DESKTOP)
  set_target_properties(arborenum-builder PROPERTIES OUTPUT_NAME AnytimeTreeBuilder)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(live-session-test tests/live_session_test.cpp)
  configure_native(live-session-test)
  add_test(NAME live-session COMMAND live-session-test)
  set_tests_properties(live-session PROPERTIES TIMEOUT 120)
endif()
install(TARGETS arborenum-builder RUNTIME DESTINATION .)
