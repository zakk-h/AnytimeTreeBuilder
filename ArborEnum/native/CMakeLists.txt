cmake_minimum_required(VERSION 3.20)
project(ArborEnumLiveBuilder LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(arborenum-builder main.cpp)
target_compile_features(arborenum-builder PRIVATE cxx_std_20)
target_link_libraries(arborenum-builder PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(arborenum-builder PRIVATE /W4 /bigobj)
  target_link_libraries(arborenum-builder PRIVATE ws2_32 psapi)
else()
  target_compile_options(arborenum-builder PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(live-session-test tests/live_session_test.cpp)
  target_compile_features(live-session-test PRIVATE cxx_std_20)
  target_link_libraries(live-session-test PRIVATE Threads::Threads)
  add_test(NAME live-session COMMAND live-session-test)
endif()
